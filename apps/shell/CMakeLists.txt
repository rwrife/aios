cmake_minimum_required(VERSION 3.21)
project(aios-shell VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Quick Network Multimedia)
qt_add_executable(aios-shell main.cpp voice.h)
qt_add_resources(aios-shell ui PREFIX / FILES Main.qml ChatWindow.qml VoiceButton.qml)
target_link_libraries(aios-shell PRIVATE Qt6::Quick Qt6::Network Qt6::Multimedia)
install(TARGETS aios-shell RUNTIME DESTINATION bin)
