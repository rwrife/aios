cmake_minimum_required(VERSION 3.21)
project(aios-shell VERSION 0.1.0 LANGUAGES CXX)
option(AIOS_EMBEDDED_DISPLAY "Build the experimental private Wayland compositor" OFF)
option(AIOS_DISPLAY_TESTS "Build the non-installed display input test harness" OFF)
option(AIOS_PROFILE_TESTS "Build the local account integration test" OFF)
set(AIOS_COMMIT "unknown" CACHE STRING "Source commit embedded in the About page")
set(AIOS_BUILD_NUMBER "development" CACHE STRING "Build number embedded in the About page")
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
configure_file(BuildInfo.h.in ${CMAKE_CURRENT_BINARY_DIR}/BuildInfo.h @ONLY)
find_package(Qt6 REQUIRED COMPONENTS Quick Network Multimedia)
if(AIOS_PROFILE_TESTS)
    qt_add_executable(aios-profile-test ../../tests/chat_profile_control.cpp CameraDevice.h SessionControl.h ProfilePhoto.h)
    target_include_directories(aios-profile-test PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
    target_link_libraries(aios-profile-test PRIVATE Qt6::Quick Qt6::Network Qt6::Multimedia)
endif()
qt_add_executable(aios-shell main.cpp voice.h CameraDevice.h SessionControl.h DisplayBridge.h ProfilePhoto.h)
qt_add_executable(aios-app-host app_host.cpp)
target_include_directories(aios-shell PRIVATE ${CMAKE_CURRENT_BINARY_DIR})
if(AIOS_EMBEDDED_DISPLAY)
    find_package(Qt6 REQUIRED COMPONENTS WaylandCompositor)
    target_sources(aios-shell PRIVATE PrivateCompositor.h)
    target_link_libraries(aios-shell PRIVATE Qt6::WaylandCompositor)
    target_compile_definitions(aios-shell PRIVATE AIOS_EMBEDDED_DISPLAY=1)
    if(AIOS_DISPLAY_TESTS)
        qt_add_executable(aios-display-input ../../tests/display_input.cpp CameraDevice.h SessionControl.h DisplayBridge.h PrivateCompositor.h ProfilePhoto.h)
        target_include_directories(aios-display-input PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
        target_compile_definitions(aios-display-input PRIVATE AIOS_EMBEDDED_DISPLAY=1)
        target_link_libraries(aios-display-input PRIVATE Qt6::WaylandCompositor Qt6::Quick Qt6::Network Qt6::Multimedia)
    endif()
endif()
qt_add_resources(aios-shell ui PREFIX / FILES Theme.qml Main.qml ChatWindow.qml VoiceButton.qml ModelSettings.qml LocalModels.qml SettingsWindow.qml SetupWizard.qml ChatOrb.qml IdentityStatus.qml PrivacyShield.qml SecurePinPrompt.qml EnrollmentFlow.qml UserBubble.qml WindowControlButton.qml WindowTitle.qml)
qt_add_resources(aios-shell protected_input PREFIX / FILES SecurePinOverlay.qml AccountSettings.qml)
qt_add_resources(aios-app-host app_host_ui PREFIX / FILES AppHost.qml)
if(AIOS_EMBEDDED_DISPLAY)
    qt_add_resources(aios-shell private_display PREFIX / FILES PrivateDisplay.qml)
endif()
target_link_libraries(aios-shell PRIVATE Qt6::Quick Qt6::Network Qt6::Multimedia)
target_link_libraries(aios-app-host PRIVATE Qt6::Quick)
install(TARGETS aios-shell aios-app-host RUNTIME DESTINATION bin)
